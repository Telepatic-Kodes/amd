**1. Update app/(dashboard)/layout.tsx:**
- Import ProductTour component
- Add ProductTour at root level (outside main content, before closing body tag)
- Render conditionally: Only show if `!getTourState().completed && !getTourState().skipped`
- Use client-side check with useEffect to prevent SSR issues

Example integration:
```tsx
"use client";

import { useState, useEffect } from "react";
import { ProductTour } from "@/components/ui/ProductTour";
import { getTourState } from "@/lib/tour-utils";

export default function DashboardLayout({ children }) {
  const [showTour, setShowTour] = useState(false);

  useEffect(() => {
    const tourState = getTourState();
    setShowTour(!tourState.completed && !tourState.skipped);
  }, []);

  return (
    <>
      {children}
      {showTour && <ProductTour onComplete={() => setShowTour(false)} />}
    </>
  );
}
```
