**Component structure:**

```typescript
"use client";

import { useState } from "react";
import { X, Upload, FileText, Check, AlertCircle } from "lucide-react";
import { motion, AnimatePresence } from "framer-motion";
import { FileDropZone } from "./FileDropZone";
import { RichTextEditor } from "./RichTextEditor";
import type { ParsedContent } from "@/lib/file-parsers";
import { cn } from "@/lib/utils";

interface FileImportModalProps {
  isOpen: boolean;
  onClose: () => void;
  onImport: (content: { title: string; body: string; metadata: any }) => void;
  defaultTitle?: string;
}

export function FileImportModal({
  isOpen,
  onClose,
  onImport,
  defaultTitle = "",
}: FileImportModalProps) {
  const [step, setStep] = useState<"upload" | "preview">("upload");
  const [parsedContent, setParsedContent] = useState<ParsedContent | null>(null);
  const [editedTitle, setEditedTitle] = useState("");
  const [editedBody, setEditedBody] = useState("");
  const [error, setError] = useState<string | null>(null);

  // Reset state when modal opens/closes
  // Handle file processed from FileDropZone
  // Handle import confirmation
}
```

**User flow:**

1. **Step 1: Upload** (step === "upload")
   - Display FileDropZone component
   - Instructions: "Arrastra un archivo PDF, DOCX o TXT para importar contenido"
   - On file processed: Move to preview step, populate editedTitle and editedBody

2. **Step 2: Preview** (step === "preview")
   - Display parsed content in RichTextEditor (read-only mode first)
   - Show file metadata (fileName, wordCount, readingTime)
   - Editable title field (pre-filled with fileName without extension)
   - Toggle "Editar" to make RichTextEditor editable
   - "Importar" button to confirm import
   - "Cancelar" button to go back to upload step

**Implementation details:**

1. **File processing:**
   ```typescript
   const handleFileProcessed = (content: ParsedContent) => {
     setParsedContent(content);
     setEditedTitle(content.metadata.fileName.replace(/\.[^/.]+$/, "")); // Remove extension
     setEditedBody(content.html); // Use HTML for TipTap
     setStep("preview");
     setError(null);
   };
   ```

2. **Preview UI:**
   ```tsx
   {step === "preview" && parsedContent && (
     <>
       <div className="mb-4">
         <FileText className="w-6 h-6 text-indigo-400" />
         <p className="text-sm text-zinc-400">
           {parsedContent.metadata.fileName} • {parsedContent.metadata.wordCount} palabras • {parsedContent.metadata.readingTime} min lectura
         </p>
       </div>

       <div className="mb-4">
         <label>Título</label>
         <input
           type="text"
           value={editedTitle}
           onChange={(e) => setEditedTitle(e.target.value)}
           className="w-full px-3 py-2 bg-zinc-800 border border-zinc-700 rounded"
         />
       </div>

       <div className="mb-4">
         <label>Contenido</label>
         <RichTextEditor
           value={editedBody}
           onChange={setEditedBody}
           placeholder="Edita el contenido importado..."
         />
       </div>

       <div className="flex gap-2">
         <button onClick={handleImport}>Importar</button>
         <button onClick={() => setStep("upload")}>Cancelar</button>
       </div>
     </>
   )}
   ```

3. **Import confirmation:**
   ```typescript
   const handleImport = () => {
     if (!editedTitle.trim()) {
       setError("El título es obligatorio");
       return;
     }
     if (editedBody.trim().length < 50) {
       setError("El contenido debe tener al menos 50 caracteres");
       return;
     }

     onImport({
       title: editedTitle,
       body: editedBody,
       metadata: parsedContent!.metadata,
     });

     // Reset and close
     setStep("upload");
     setParsedContent(null);
     setEditedTitle("");
     setEditedBody("");
     onClose();
   };
   ```

4. **Error handling:**
   - Display error from FileDropZone (parsing errors)
   - Display validation errors (empty title, short content)
   - Retry option: Allow user to go back to upload step

**Styling:**
- Full-screen modal with dark background overlay
- Max width 4xl for comfortable editing

- Mobile-responsive (single column on small screens)
- Touch-friendly buttons (44x44px minimum)

```tsx
      >
        {/* Modal content */}

    </motion.div>
  )}

```

**IMPORTANT - HTML Sanitization:**
The parsed HTML content is already controlled by our file parsers (pdf-parse and mammoth), which generate clean HTML. TipTap's RichTextEditor component also sanitizes HTML through its schema. However, for defense-in-depth:

- File parsers generate HTML from trusted sources (PDF/DOCX structure, not user input)
- RichTextEditor (TipTap) enforces its HTML schema and sanitizes on render
- Content is displayed only through RichTextEditor, which is safe

No additional sanitization needed because:
1. Source is not user-provided HTML (it's generated from document structure)
2. TipTap provides schema-based sanitization
3. Content never rendered with dangerouslySetInnerHTML in this component
