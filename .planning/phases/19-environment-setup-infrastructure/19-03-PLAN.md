1. Create `.github/workflows/ci.yml` for PR checks:

```yaml
name: CI

on:
  pull_request:
    branches: [main]

jobs:
  lint-and-typecheck:
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: ai-marketing-department/ai-marketing-department
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: 'npm'
          cache-dependency-path: ai-marketing-department/ai-marketing-department/package-lock.json
      - run: npm ci
      - name: Lint
        run: npm run lint
      - name: Typecheck
        run: npx tsc --noEmit
      - name: Check for leaked secrets in NEXT_PUBLIC_ vars
        run: |
          # CICD-05: Fail if any NEXT_PUBLIC_ env var name contains KEY or SECRET
          if grep -rn 'NEXT_PUBLIC_.*\(KEY\|SECRET\)' --include='*.ts' --include='*.tsx' --include='*.env*' . ../convex/ 2>/dev/null | grep -v node_modules | grep -v '.env.example' | grep -v '.env.local'; then
            echo "ERROR: Found NEXT_PUBLIC_ variable containing KEY or SECRET - this would expose secrets to the client!"
            exit 1
          fi
          echo "No leaked secrets detected in NEXT_PUBLIC_ variables"

  test:
    runs-on: ubuntu-latest
    defaults:
      run:
        working-directory: ai-marketing-department/ai-marketing-department
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: 'npm'
          cache-dependency-path: ai-marketing-department/ai-marketing-department/package-lock.json
      - run: npm ci
      - name: Test
        run: npm test -- --run
        # --run flag makes Vitest exit after tests complete (non-watch mode)
```

NOTES:
- Working directory set to frontend because that's where lint/typecheck/tests run
- `npm ci` for reproducible installs
- Node 20 (LTS)
- Two jobs (lint-and-typecheck + test) run in parallel for speed
- The secret leak check (CICD-05) scans for NEXT_PUBLIC_ vars that contain KEY or SECRET in the name -- this would expose server secrets to the browser
- Test job uses `--run` flag for Vitest non-watch mode

1. Create `.github/workflows/deploy.yml` for production deploys:

```yaml
name: Deploy

on:
  push:
    branches: [main]

jobs:
  deploy:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      # Step 1: Deploy Convex (must happen FIRST - generates types for Next.js build)
      - uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: 'npm'
          cache-dependency-path: package-lock.json
      - name: Install Convex dependencies
        run: npm ci
      - name: Deploy Convex to production
        run: npx convex deploy --prod
        env:
          CONVEX_DEPLOY_KEY: ${{ secrets.CONVEX_DEPLOY_KEY }}

      # Step 2: Deploy Next.js to Vercel (after Convex types are ready)
      - name: Install Vercel CLI
        run: npm install -g vercel@latest
      - name: Pull Vercel Environment
        run: vercel pull --yes --environment=production --token=${{ secrets.VERCEL_TOKEN }}
        working-directory: ai-marketing-department/ai-marketing-department
      - name: Build
        run: vercel build --prod --token=${{ secrets.VERCEL_TOKEN }}
        working-directory: ai-marketing-department/ai-marketing-department
      - name: Deploy to Vercel
        run: vercel deploy --prebuilt --prod --token=${{ secrets.VERCEL_TOKEN }}
        working-directory: ai-marketing-department/ai-marketing-department
```

CRITICAL SEQUENCING: Convex must deploy FIRST because `npx convex deploy` generates TypeScript types that the Next.js build depends on. If you deploy Vercel first, the build may use stale Convex types.

Required GitHub Secrets (user must set these):
- `CONVEX_DEPLOY_KEY`: From Convex Dashboard -> Settings -> Deploy Key
- `VERCEL_TOKEN`: From Vercel Dashboard -> Settings -> Tokens
- `VERCEL_ORG_ID`: From Vercel project settings (auto-pulled)
- `VERCEL_PROJECT_ID`: From Vercel project settings (auto-pulled)
