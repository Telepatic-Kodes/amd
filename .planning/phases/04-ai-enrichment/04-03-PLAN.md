Create `convex/enrichment/orchestration.ts` with:

**processBatch** - action (public, called by cron)
- Args: `{ batchSize: v.optional(v.number()) }`
- Default batchSize: 10
- Returns: `{ processed: number, failed: number, skipped: number }`

Implementation:
```typescript
import { v } from "convex/values";
import { action } from "../_generated/server";
import { internal } from "../_generated/api";

/**
 * Process a batch of unprocessed feed items through AI enrichment.
 *
 * Called by cron job to incrementally enrich feed items.
 * Processes items sequentially to avoid rate limits.
 *
 * @param batchSize - Number of items to process (default: 10)
 * @returns Counts of processed, failed, and skipped items
 */
export const processBatch = action({
  args: {
    batchSize: v.optional(v.number()),
  },
  handler: async (ctx, args): Promise<{
    processed: number;
    failed: number;
    skipped: number;
    totalTokens: number;
  }> => {
    const batchSize = args.batchSize ?? 10;

    // 1. Get unprocessed items (limit to batch size)
    const items = await ctx.runQuery(
      internal.enrichment.queries.getUnprocessedItems,
      { limit: batchSize }
    );

    if (items.length === 0) {
      console.log("[enrichment] No unprocessed items found");
      return { processed: 0, failed: 0, skipped: 0, totalTokens: 0 };
    }

    console.log(`[enrichment] Processing batch of ${items.length} items`);

    let processed = 0;
    let failed = 0;
    let skipped = 0;
    let totalTokens = 0;

    // 2. Process items sequentially (avoid rate limits)
    for (const item of items) {
      // Skip items that are already marked as failed (processed: false)
      // These have processingError set and should not retry automatically
      if (item.processed === false) {
        skipped++;
        continue;
      }

      try {
        const result = await ctx.runAction(
          internal.enrichment.processItems.enrichFeedItem,
          { itemId: item._id }
        );
        processed++;
        totalTokens += result.tokensUsed;
      } catch (error: any) {
        console.error(
          `[enrichment] Failed to enrich item ${item._id}: ${error.message}`
        );

        // Mark as failed to prevent infinite retries
        await ctx.runMutation(internal.enrichment.mutations.markFailed, {
          itemId: item._id,
          error: error.message || "Unknown error",
        });

        failed++;
      }
    }

    console.log(
      `[enrichment] Batch complete: ${processed} processed, ${failed} failed, ${skipped} skipped, ${totalTokens} tokens`
    );

    return { processed, failed, skipped, totalTokens };
  },
});

/**
 * Manually trigger enrichment processing (for testing/admin).
 *
 * Same as processBatch but exposed for manual invocation.
 */
export const triggerEnrichment = action({
  args: {
    batchSize: v.optional(v.number()),
  },
  handler: async (ctx, args) => {
    return processBatch.handler(ctx, args);
  },
});
```

Key implementation notes:
- Sequential processing avoids Claude API rate limits
- Failed items marked with `processed: false` to prevent infinite retries
- Skips already-failed items (they have processingError set)
- Logs progress for monitoring
- Returns token count for cost tracking
- triggerEnrichment provides manual invocation for testing

```typescript
// ===========================================
// AI ENRICHMENT CRONS

// Process unprocessed feed items through AI enrichment
// Runs 30 minutes after daily feed sync to allow items to populate
// Processes 10 items per run to control costs
crons.daily(
  "enrich-feed-items",
  { hourUTC: 6, minuteUTC: 30 }, // 30 min after daily sync at 6:00
  api.enrichment.orchestration.processBatch,
  { batchSize: 10 }
);

// Hourly enrichment catch-up for newly synced items
// Processes smaller batch (5) more frequently
crons.hourly(
  "enrich-feed-items-hourly",
  { minuteUTC: 35 }, // 30 min after hourly sync at :05
  api.enrichment.orchestration.processBatch,
  { batchSize: 5 }
);
```

Place this section after the "FEED SYNC CRONS" section and before `export default crons;`.

The timing ensures:
- Daily: 30 min after 6:00 feed sync (6:30 UTC)
- Hourly: 30 min after :05 hourly sync (:35)
- This gap allows feed items to be stored before enrichment runs
